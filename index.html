// AEGIS-PRIME Serial Relay Bridge (ESP32 / Arduino)
// Baud: 115200 | Hardware-in-the-Loop Controller

const int PIN_RELAY_TRANS = 4;   // RLY-01: HV Transformer
const int PIN_RELAY_CRYO  = 5;   // RLY-02: Cryo Pump Bypass
const int PIN_RELAY_RF    = 18;  // RLY-03: RF Attenuator
const int PIN_RELAY_CLOCK = 19;  // RLY-04: Inertial Holdover
const int PIN_LED_ALERT   = 2;   // Status Indicator

void setup() {
  Serial.begin(115200);
  pinMode(PIN_RELAY_TRANS, OUTPUT);
  pinMode(PIN_RELAY_CRYO, OUTPUT);
  pinMode(PIN_RELAY_RF, OUTPUT);
  pinMode(PIN_RELAY_CLOCK, OUTPUT);
  pinMode(PIN_LED_ALERT, OUTPUT);
  digitalWrite(PIN_RELAY_TRANS, HIGH); // Closed (Normal)
  Serial.println("AEGIS_HIL_ONLINE:READY");
}

void loop() {
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd == "CMD:TRIP_RELAY_01") {
      digitalWrite(PIN_RELAY_TRANS, LOW); // Open Breaker
      digitalWrite(PIN_LED_ALERT, HIGH);
      Serial.println("ACK:RLY-01_TRIPPED:ISOLATED");
    } else if (cmd == "CMD:ENGAGE_CRYO_PULSE") {
      digitalWrite(PIN_RELAY_CRYO, LOW);
      Serial.println("ACK:RLY-02_PULSE_ACTIVE");
    } else if (cmd == "CMD:SYNC_ALL_NOMINAL") {
      digitalWrite(PIN_RELAY_TRANS, HIGH);
      digitalWrite(PIN_LED_ALERT, LOW);
      Serial.println("ACK:ALL_RELAYS_NOMINAL");
    }
  }
}
